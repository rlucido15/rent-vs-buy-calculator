/** * Lofty lead intake — a SHARED Google Apps Script Web App. * =========================================================================== * ONE endpoint that every Claude-built CTA / landing page / form can POST leads * to. It holds your Lofty API token server-side, then for each incoming lead it: * * 1. SEARCHES Lofty for an existing lead (by email, then phone). * 2. If found -> UPDATES that lead (adds the CTA tag, fills blanks, pins a * fresh note with the new submission). * If not -> CREATES a new lead. * * It is CTA-agnostic. Any form can post to it. Whatever standard contact fields * it recognizes (name, email, phone, address...) are mapped to the right Lofty * fields; every OTHER field on the form is preserved (in a readable pinned note, * and — when you configure them — as Lofty custom fields). The lead's tag is * ALWAYS taken from the payload and always stored as "CTA: ". * * ========================= ONE-TIME SETUP (do this once) ========================= * 1. script.google.com -> New project. Paste this whole file in. Save. * 2. Project Settings (gear) -> Script properties -> Add property: * LOFTY_TOKEN = your Lofty API key (Settings > Integrations > API in Lofty) * Optionally also add: * SHARED_SECRET = any long random string (a light spam guard). If you set * this, every CTA must send the same value as `secret`. * Leave it unset to disable the check. * 3. Deploy -> New deployment -> "Web app". * Execute as: Me * Who has access: Anyone <-- REQUIRED, or browsers get "Failed to fetch" * Copy the /exec URL. That URL goes in every CTA's LEAD_ENDPOINT. * 4. Run testPing() once from the editor to authorize outbound fetch (Google will * prompt the first time), and to push a real test lead into Lofty. * * ========================= ADDING A NEW CTA LATER ========================= * Nothing to change here. Point the new page at the same /exec URL and POST a * payload like: * * { * "name": "Jane Smith", // or firstName / lastName * "email": "jane@email.com", * "phone": "5551234567", * "cta": "Rent vs. Buy Calculator", // becomes tag "CTA: Rent vs. Buy Calculator" * "source": "MoxieMortgage.com", * "consent": true, // marks the lead contactable * "secret": "<SHARED_SECRET if you set one>", * "fields": { // OPTIONAL: any extra form fields, freeform * "Home Price": "$400,000", * "Down Payment": "10%", * "Credit Score": 740 * } * } * * `cta` (or `tag`) drives the tag. `fields` (or `inputs` / `results`) is any * key/value data your form collected — it is preserved on the lead automatically. * * SECURITY: the Lofty token + shared secret live only in Script Properties, never * in any page's client code, so the CTAs are safe to embed publicly. */ var LOFTY_BASE = 'https://api.lofty.com/v1.0/leads'; /* =============================== HTTP entry =============================== */ function doPost(e) { try { // Accept the payload as a "payload" form field (hidden-iframe method, the // reliable way to reach Apps Script) OR as a raw JSON body (fetch). var raw = (e && e.parameter && e.parameter.payload) ? e.parameter.payload : ((e && e.postData && e.postData.contents) || '{}'); var body = JSON.parse(raw); // Optional shared-secret gate. var expected = PropertiesService.getScriptProperties().getProperty('SHARED_SECRET'); if (expected && String(body.secret || '') !== expected) { return json({ ok: false, error: 'unauthorized' }); } var token = PropertiesService.getScriptProperties().getProperty('LOFTY_TOKEN'); if (!token) return json({ ok: false, error: 'server not configured (LOFTY_TOKEN missing)' }); var email = String(body.email || '').trim(); var phone = toE164(String(body.phone || '')); var name = splitName(String(body.firstName || body.name || '') + (body.lastName ? ' ' + body.lastName : '')); if (!name.firstName && !email && !phone) { return json({ ok: false, error: 'a name, email, or phone is required' }); } // 1) Search Lofty for an existing lead (email first, then phone). var existingId = findExistingLeadId(token, email, phone); // 2) Update if found, else create. if (existingId) { var upd = buildUpdatePayload(body, name, email, phone); var r1 = loftyFetch(token, 'put', LOFTY_BASE + '/' + existingId, upd); if (r1.ok) return json({ ok: true, action: 'updated', leadId: existingId, lofty: r1.data }); return json({ ok: false, action: 'update', error: 'lofty ' + r1.code, detail: r1.text }); } else { var crt = buildCreatePayload(body, name, email, phone); var r2 = loftyFetch(token, 'post', LOFTY_BASE, crt); if (r2.ok) return json({ ok: true, action: 'created', lofty: r2.data }); return json({ ok: false, action: 'create', error: 'lofty ' + r2.code, detail: r2.text }); } } catch (err) { return json({ ok: false, error: String(err) }); } } // GET just confirms the endpoint is alive (open the /exec URL in a browser). function doGet() { return json({ ok: true, service: 'lofty-lead-intake', ready: true }); } /* ============================ Lofty search ============================ */ /** * Returns the leadId of an existing Lofty lead matching this email or phone, * or null if none is found. Email is tried first (most reliable), then phone. */ function findExistingLeadId(token, email, phone) { function search(param, value) { if (!value) return null; var url = LOFTY_BASE + '?preciseSearchFlag=true&limit=1&' + param + '=' + encodeURIComponent(value); var r = loftyFetch(token, 'get', url, null); if (!r.ok || !r.data) return null; var leads = r.data.leads || r.data.data || []; if (leads.length && (leads[0].leadId || leads[0].id)) { return leads[0].leadId || leads[0].id; } return null; } return search('email', email) || search('phone', phone && phone.replace(/^\+/, '')); } /* ==================== Build create / update payloads ==================== */ /** Fields common to both create and update. */ function baseLeadFields(body, name, email, phone) { var contactable = body.consent === true; var lead = { firstName: name.firstName, emails: email ? [email] : [], phones: phone ? [phone] : [], leadTypes: [2], // Buyer source: String(body.source || 'MoxieMortgage.com'), cannotCall: !contactable, cannotText: !contactable, cannotEmail: !contactable }; if (name.lastName) lead.lastName = name.lastName; // Address / property (any of these the form supplied). var prop = {}; var city = str(body.city), state = str(body.state); var zip = str(body.zip || body.zipCode), street = str(body.streetAddress || body.address); if (city) prop.city = city; if (state) prop.state = state; if (zip) prop.zipCode = zip; if (street) prop.streetAddress = street; if (Object.keys(prop).length) lead.property = prop; // Common mortgage/real-estate flags, only if the form sent them. var tf = TIMEFRAME[String(body.purchaseTimeline || body.timeline || '')]; if (tf) lead.buyingTimeFrame = tf; if (str(body.preQual)) lead.preQual = yesNo(body.preQual); if (str(body.firstTimeBuyer || body.fthb)) lead.fthb = yesNo(body.firstTimeBuyer || body.fthb); if (body.homePrice != null || body.mortgage) lead.mortgage = 'Yes'; return lead; } /** The tag for this lead, always shaped "CTA: <Title>". */ function ctaTag(body) { var raw = String(body.cta || body.tag || '').trim(); if (!raw) raw = 'General'; // normalize: strip an existing "CTA:" prefix so we don't double it raw = raw.replace(/^\s*CTA:\s*/i, ''); return trim64('CTA: ' + raw); } /** New-lead payload (POST /v1.0/leads). */ function buildCreatePayload(body, name, email, phone) { var lead = baseLeadFields(body, name, email, phone); lead.tags = [ctaTag(body)]; var note = buildNote(body, name, email, phone); if (note) { lead.content = note; lead.isPin = true; } var custom = buildCustomAttributes(body); if (custom.length) lead.customAttributeList = custom; return lead; } /** Existing-lead payload (PUT /v1.0/leads/{id}). * Uses tagsAdd so we KEEP the lead's existing tags and just add this CTA's tag. * Re-pins a note with the latest submission so the agent sees the newest intent. */ function buildUpdatePayload(body, name, email, phone) { var lead = baseLeadFields(body, name, email, phone); lead.tagsAdd = [ctaTag(body)]; // additive — never wipes existing tags var note = buildNote(body, name, email, phone); if (note) { lead.content = note; lead.isPin = true; } var custom = buildCustomAttributes(body); if (custom.length) lead.customAttributeList = custom; return lead; } /** * Turns arbitrary extra form data into Lofty custom-field values. Reads from * body.fields, body.inputs and body.results (any/all may be present). These only * land in Lofty if you've created matching custom fields on your team AND you map * them here by name; otherwise the same data is still captured in the pinned note * below, so nothing is lost. Left as a safe pass-through: only include entries * whose attributeName you know exists as a lead custom field in your Lofty team. * * To activate: fill CUSTOM_FIELD_MAP with { payloadKey: "Lofty Custom Field Name" }. */ var CUSTOM_FIELD_MAP = { // Example (uncomment + match to your Lofty custom fields): // homePrice: 'Home Price', // downPayment: 'Down Payment', // creditScore: 'Credit Score' }; function buildCustomAttributes(body) { var out = []; var pools = [body.fields, body.inputs, body.results]; for (var p = 0; p < pools.length; p++) { var pool = pools[p]; if (!pool || typeof pool !== 'object') continue; for (var key in pool) { if (!pool.hasOwnProperty(key)) continue; var mapped = CUSTOM_FIELD_MAP[key]; if (!mapped) continue; // only send fields you've mapped to real Lofty custom fields out.push({ attributeName: mapped, attributeType: 'text', value: String(pool[key]) }); } } return out; } /* =============================== Notes =============================== */ /** * A readable pinned note capturing the whole submission, so no form field is * ever lost even if you haven't set up custom fields for it. Sections: contact, * then every key from fields/inputs/results, then meta. */ function buildNote(body, name, email, phone) { var lines = []; var title = String(body.cta || body.tag || 'Lead').replace(/^\s*CTA:\s*/i, ''); lines.push(title.toUpperCase() + ' — SUBMISSION'); lines.push(''); lines.push('CONTACT'); if (name.firstName || name.lastName) lines.push('Name: ' + [name.firstName, name.lastName].filter(Boolean).join(' ')); if (email) lines.push('Email: ' + email); if (phone) lines.push('Phone: ' + phone); var loc = [str(body.streetAddress || body.address), str(body.city), str(body.state), str(body.zip || body.zipCode)].filter(Boolean).join(', '); if (loc) lines.push('Location: ' + loc); appendPool(lines, 'DETAILS', body.fields); appendPool(lines, 'INPUTS', body.inputs); appendPool(lines, 'RESULTS', body.results); // A couple of well-known single fields some forms send flat. var flat = []; if (body.message) flat.push('Message: ' + String(body.message)); if (body.homePrice != null) flat.push('Home Price: ' + body.homePrice); if (flat.length) { lines.push(''); lines.push('MORE'); lines.push.apply(lines, flat); } lines.push(''); lines.push('Source: ' + String(body.source || 'MoxieMortgage.com')); if (body.pageUrl) lines.push('Page: ' + String(body.pageUrl)); lines.push('Submitted: ' + formatWhen(body.submittedAt ? new Date(body.submittedAt) : new Date())); return lines.join('\n').slice(0, 4000); } function appendPool(lines, heading, pool) { if (!pool || typeof pool !== 'object') return; var keys = Object.keys(pool); if (!keys.length) return; lines.push(''); lines.push(heading); for (var i = 0; i < keys.length; i++) { lines.push(prettyKey(keys[i]) + ': ' + String(pool[keys[i]])); } } /* ============================== Lofty fetch ============================== */ /** * Single place that talks to Lofty. Returns { ok, code, text, data }. * Lofty API keys authenticate as "token <key>" (NOT "Bearer <key>"). */ function loftyFetch(token, method, url, payloadObj) { var opts = { method: method, headers: { Authorization: 'token ' + token }, muteHttpExceptions: true }; if (payloadObj != null) { opts.contentType = 'application/json'; opts.payload = JSON.stringify(payloadObj); } var res = UrlFetchApp.fetch(url, opts); var code = res.getResponseCode(); var text = res.getContentText(); return { ok: code >= 200 && code < 300, code: code, text: text, data: safeParse(text) }; } /* ============================== Helpers ============================== */ var TIMEFRAME = { 'buying-now': '1-3', '0-3-months': '1-3', '3-6-months': '3-6', '6-plus-months': '6-12', 'researching': 'Just Looking', 'refinancing': 'Refinancing' }; function str(v) { return v == null ? '' : String(v).trim(); } function yesNo(v) { var s = String(v).trim().toLowerCase(); return (s === 'yes' || s === 'true' || v === true) ? 'Yes' : 'No'; } function trim30(s) { return String(s || '').slice(0, 30); } function trim64(s) { return String(s || '').slice(0, 64); } function splitName(raw) { var full = String(raw || '').trim().replace(/\s+/g, ' '); if (!full) return { firstName: '' }; var parts = full.split(' '); if (parts.length === 1) return { firstName: trim30(parts[0]) }; return { firstName: trim30(parts[0]), lastName: trim30(parts.slice(1).join(' ')) }; } function toE164(phone) { var digits = String(phone || '').replace(/[^\d]/g, ''); if (String(phone).trim().charAt(0) === '+') return '+' + digits; if (digits.length === 10) return '+1' + digits; if (digits.length === 11 && digits.charAt(0) === '1') return '+' + digits; return digits ? '+' + digits : ''; } function prettyKey(k) { return String(k) .replace(/([a-z])([A-Z])/g, '$1 $2') .replace(/[_-]+/g, ' ') .replace(/\b\w/g, function (c) { return c.toUpperCase(); }); } function formatWhen(d) { try { if (typeof Utilities !== 'undefined' && Utilities.formatDate) { var tz = (typeof Session !== 'undefined' && Session.getScriptTimeZone) ? Session.getScriptTimeZone() : 'America/New_York'; return Utilities.formatDate(d, tz, 'M/d/yyyy h:mm a'); } } catch (e) {} try { return d.toISOString().replace('T', ' ').slice(0, 16) + ' UTC'; } catch (e2) { return String(d); } } function safeParse(text) { try { return JSON.parse(text); } catch (e) { return text; } } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* ============================== Test ============================== */ // Run once from the editor to authorize outbound fetch and push a real test // lead into Lofty. Run it twice to confirm the 2nd run UPDATES (not duplicates). function testPing() { var secret = PropertiesService.getScriptProperties().getProperty('SHARED_SECRET') || ''; var fake = { postData: { contents: JSON.stringify({ secret: secret, name: 'Test Lead', email: 'test-lead@example.com', phone: '4104018688', zip: '30301', consent: true, cta: 'Rent vs. Buy Calculator', source: 'MoxieMortgage.com', fields: { 'Home Price': '$400,000', 'Down Payment': '10%', 'Interest Rate': '6.75%' }, results: { 'Net Worth Advantage': '$1,093,171', 'Tax Savings': '$156,965' }, submittedAt: new Date().toISOString(), pageUrl: 'https://moxiemortgage.com/rent-vs-buy' }) } }; Logger.log(doPost(fake).getContent()); }
